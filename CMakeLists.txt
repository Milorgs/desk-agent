cmake_minimum_required(VERSION 3.24)
project(desk_agent VERSION 1.6.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(json-c CONFIG REQUIRED)
if(TARGET json-c::json-c)
  set(JSON_TARGET json-c::json-c)
else()
  set(JSON_TARGET json-c::json-c-static)
endif()
set(DESK_COMMON src/mcp.cpp src/common.cpp src/platform.cpp src/config.cpp src/api.cpp src/files.cpp src/agent.cpp src/i18n.cpp src/tasks_ui.cpp)
if(WIN32)
  set(DESK_PLATFORM src/windows/util.cpp src/windows/process.cpp src/windows/tasks.cpp src/windows/desktop.cpp)
else()
  set(DESK_PLATFORM src/util.cpp src/process.cpp src/tasks.cpp src/desktop.cpp)
endif()
add_library(desk_core STATIC ${DESK_COMMON} ${DESK_PLATFORM})
target_include_directories(desk_core PUBLIC src)
target_link_libraries(desk_core PUBLIC CURL::libcurl OpenSSL::SSL OpenSSL::Crypto ${JSON_TARGET})
if(WIN32)
  target_compile_definitions(desk_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(desk_core PUBLIC advapi32 ole32 oleaut32 uiautomationcore ws2_32)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(READLINE REQUIRED IMPORTED_TARGET readline)
  target_link_libraries(desk_core PUBLIC PkgConfig::READLINE)
  pkg_check_modules(ATSPI IMPORTED_TARGET atspi-2 gobject-2.0)
  if(ATSPI_FOUND)
    target_compile_definitions(desk_core PRIVATE WITH_ATSPI)
    target_link_libraries(desk_core PUBLIC PkgConfig::ATSPI)
  endif()
  target_link_libraries(desk_core PUBLIC util)
endif()
add_executable(desk src/main.cpp)
target_link_libraries(desk PRIVATE desk_core)
if(MSVC)
  target_compile_options(desk_core PRIVATE /W4 /utf-8 /EHsc /permissive-)
  target_compile_options(desk PRIVATE /W4 /utf-8 /EHsc /permissive-)
  target_link_options(desk PRIVATE "/STACK:8388608" "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows/desk.manifest")
else()
  target_compile_options(desk_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(desk PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  if(WIN32)
    add_executable(desk_tests tests/windows.cpp)
    target_compile_options(desk_tests PRIVATE /W4 /utf-8 /EHsc /permissive-)
    target_link_options(desk_tests PRIVATE /STACK:8388608)
    target_link_libraries(desk_tests PRIVATE desk_core)
    add_test(NAME windows_core COMMAND desk_tests $<TARGET_FILE:desk>)
    set_tests_properties(windows_core PROPERTIES TIMEOUT 120)
  else()
    add_executable(desk_tests tests/test.cpp)
    target_link_libraries(desk_tests PRIVATE desk_core)
    # Legacy integration tests exec ./build/desk from the repository root.
    add_test(NAME linux_core COMMAND desk_tests)
    set_tests_properties(linux_core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 120)
  endif()
endif()
install(TARGETS desk RUNTIME DESTINATION bin)
